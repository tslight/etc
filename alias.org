* rclone

#+begin_src shell
  alias dropbox_sync_to_device='rclone sync dropbox: ~/db --progress'
  alias dropbox_sync_from_device='rclone sync ~/db dropbox: --progress'
  dropbox_mount() {
      [ -z $1 ] && MAXSIZE="10G" || MAXSIZE=$1
      /usr/bin/rclone mount dropbox: "$HOME/db" \
  		    --vfs-cache-mode full \
  		    --vfs-cache-max-size "$MAXSIZE" \
  		    --vfs-cache-max-age 900h \
  		    --vfs-read-ahead 128M \
  		    --buffer-size 64M \
  		    --dir-cache-time 1m \
  		    --poll-interval 30s \
  		    --log-level NOTICE
  }
  alias dropbox_unmount='fusermount3 -u $HOME/db'
#+end_src
