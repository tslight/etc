#+TITLE: Rclone Dropbox VFS Mount Configuration
#+PROPERTY: header-args :mkdirp yes
#+PROPERTY: header-args+ :results silent
#+BEGIN_SRC ini :tangle ~/.config/systemd/user/rclone-dropbox.service
  [Unit]
  Description=Rclone Dropbox Mount (50GB Capped Profile)
  Documentation=man:rclone(1)
  After=network-online.target
  Wants=network-online.target

  [Service]
  Type=notify
  ExecStart=/usr/bin/rclone mount dropbox: %h/db \
    --vfs-cache-mode full \
    --vfs-cache-max-size 50G \
    --vfs-cache-max-age 900h \
    --vfs-read-ahead 128M \
    --buffer-size 64M \
    --dir-cache-time 1m \
    --poll-interval 30s \
    --log-level NOTICE

  ExecStop=/usr/bin/fusermount3 -u %h/db
  Restart=on-failure
  RestartSec=5

  [Install]
  WantedBy=default.target
#+END_SRC
#+BEGIN_SRC bash :tangle no
mkdir -p ~/db
systemctl --user daemon-reload
systemctl --user enable --now rclone-dropbox.service
#+END_SRC
