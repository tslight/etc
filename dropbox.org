#+TITLE: Rclone Dropbox Configuration
#+PROPERTY: header-args :mkdirp yes

* Unit files

** Limited Persistence

#+BEGIN_SRC ini :tangle ~/.config/systemd/user/rclone-dropbox.service
  [Unit]
  Description=Rclone Dropbox Mount (50GB Capped Profile)
  Documentation=man:rclone(1)
  After=network-online.target
  Wants=network-online.target

  [Service]
  Type=notify
  ExecStart=/usr/bin/rclone mount dropbox: %h/db \
    --vfs-cache-mode full \
    --vfs-cache-max-size 50G \
    --vfs-cache-max-age 900h \
    --vfs-read-ahead 128M \
    --buffer-size 64M \
    --log-level NOTICE

  ExecStop=/usr/bin/fusermount3 -u %h/db
  Restart=on-failure
  RestartSec=5

  [Install]
  WantedBy=default.target
#+END_SRC

** Unlimited Persistence

#+BEGIN_SRC ini :tangle ~/.config/systemd/user/rclone-dropbox-mirror.service
  [Unit]
  Description=Rclone Dropbox Mount (Full Mirror Profile)
  Documentation=man:rclone(1)
  After=network-online.target
  Wants=network-online.target

  [Service]
  Type=notify
  ExecStart=/usr/bin/rclone mount dropbox: %h/db \
    --vfs-cache-mode full \
    --vfs-cache-max-size off \
    --vfs-cache-max-age off \
    --vfs-read-ahead 128M \
    --buffer-size 64M \
    --log-level NOTICE

  ExecStop=/usr/bin/fusermount3 -u %h/db
  Restart=on-failure
  RestartSec=5

  [Install]
  WantedBy=default.target
#+END_SRC

* Mount via systemd

#+BEGIN_SRC bash :tangle no
mkdir -p ~/db
systemctl --user daemon-reload
systemctl --user enable --now rclone-dropbox.service
systemctl --user status rclone-dropbox.service
#+END_SRC
